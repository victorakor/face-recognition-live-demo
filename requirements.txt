# Web
Flask==3.0.3
gunicorn==22.0.0

# Numerics and imaging. numpy is held at 1.26.x because it is the version every wheel here
# is known-good against; nothing in this pinset needs numpy 2.
numpy==1.26.4
opencv-python-headless==4.10.0.84

# Detection. The YOLOv8 weights run through onnxruntime rather than torch + ultralytics:
# identical output (verified to within a pixel), ~1.6x faster on CPU, and ~80 MB resident
# instead of ~400 MB -- which is the difference between fitting a 512 MB container and not.
# detector.py does the letterboxing and NMS that ultralytics would otherwise do.
onnxruntime==1.29.0

# Face landmarks + 128-d embeddings.
#
# dlib-bin is a prebuilt wheel, so a local checkout needs no compiler. It is built without
# BLAS though, which makes the ResNet encoder roughly 20x slower (~1.5 s per face instead
# of ~80 ms). The Dockerfile replaces it with dlib compiled against OpenBLAS; the app
# reports which one it ended up with under "Model notes" on the page.
dlib-bin==19.24.6
